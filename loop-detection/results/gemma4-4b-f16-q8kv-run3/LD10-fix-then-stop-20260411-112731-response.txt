import sys

input_path = \"/tmp/ageval/ld10/input.csv\"
output_path = \"/tmp/ageval/ld10/output.txt\"

values = []
try:
    with open(input_path, mode='r', newline='') as csvfile:
        reader = csv.DictReader(csvfile)
        for row in reader:
            try:
                values.append(float(row['value']))
            except ValueError:
                # Skip rows where 'value' is not a valid float
                continue
except FileNotFoundError:
    sys.stderr.write(f\"Error: Input file not found at {input_path}\\n\")
    sys.exit(1)

if not values:
    mean_val = 0.0
else:
    mean_val = sum(values) / len(values)

with open(output_path, \"w\") as f:
    f.write(str(mean_val))
" > /tmp/ageval/ld10/process.py
