```bash
cat > /tmp/ageval/ld10/process.py << 'EOF'
import csv
import sys

input_path = "/tmp/ageval/ld10/input.csv"
output_path = "/tmp/ageval/ld10/output.txt"

values = []
with open(input_path, 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        values.append(float(row['value']))

mean_val = sum(values) / len(values)

with open(output_path, 'w') as f:
    f.write(str(mean_val))
EOF
```
