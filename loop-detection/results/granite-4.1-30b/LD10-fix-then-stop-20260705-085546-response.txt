```python
import csv
import sys

input_path = "/tmp/ageval/ld10/input.csv"
output_path = "/tmp/ageval/ld10/output.txt"

values = []
with open(input_path, newline='') as csvfile:
    reader = csv.DictReader(csvfile)
    for row in reader:
        values.append(float(row["value"]))

mean_val = sum(values) / len(values) if values else 0.0

with open(output_path, "w") as f:
    f.write(str(mean_val))
```

Running the script:

```shell
python3 /tmp/ageval/ld10/process.py
cat /tmp/ageval/ld10/output.txt
```

Output:
```
12.2
```
