```bash
# Step 1: List all repositories in the FactusConsulting organization
repos=$(gh repo list FactusConsulting --limit 50 --json name,defaultBranchRef)

# Step 2: Process each repository one by one
summary="repo name | branch name | linked issue/PR (or \"none\")\n"
summary+="--- | --- | ---\n"

jq -r '.[] | "\(.name) \(.defaultBranchRef.name)"' <<< "$repos" | while read -r repo default_branch; do
    # Get branches for the current repository
    branches=$(gh repo view FactusConsulting/"$repo" --json refs)
    branch_count=$(jq '.refs | length' <<< "$branches")
    
    # Proceed only if the repository has more than 1 branch
    if [ "$branch_count" -gt 1 ]; then
        # Extract branch names, excluding the default branch (e.g., main)
        jq -r --arg db "$default_branch" '.refs[] | select(.name != $db) | .name' <<< "$branches" | while read -r branch; do
            # Check for open issues related to the branch
            issues=$(gh issue list --repo FactusConsulting/"$repo" --state open --json number,title --limit 1 --jq ".[] | select(.title | test(\"$branch\"))")
            issue_found=$(jq -s 'length' <<< "$issues")
            
            # Check for open PRs related to the branch
            prs=$(gh pr list --repo FactusConsulting/"$repo" --state open --json number,title --limit 1 --jq ".[] | select(.title | test(\"$branch\"))")
            pr_found=$(jq -s 'length' <<< "$prs")
            
            # Determine linked issue/PR or "none"
            if [ "$issue_found" -gt 0 ]; then
                linked="issue $(jq -r '.[0].number' <<< "$issues")"
            elif [ "$pr_found" -gt 0 ]; then
                linked="PR $(jq -r '.[0].number' <<< "$prs")"
            else
                linked="none"
            fi
            
            # Append to summary table
            summary+="$repo | $branch | $linked\n"
        done
    fi
done

# Output the final summary table once
echo -e "$summary"
```
