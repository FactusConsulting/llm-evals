```bash
# Step 1: List all repositories in FactusConsulting
REPOS=$(gh repo list FactusConsulting --limit 50 --json name,defaultBranchRef | jq -r '.[] | .name')

SUMMARY_DATA=""

for REPO_NAME in $REPOS; do
    REPO_FULL_NAME="FactusConsulting/${REPO_NAME}"
    
    # Step 2a: List all branches for the current repo
    BRANCHES_JSON=$(gh repo view $REPO_FULL_NAME --json refs)
    BRANCHES=$(echo $BRANCHES_JSON | jq -r '.refs[] | select(.ref | startswith("refs/heads/")) | .ref | sub("refs/heads/"; "")')
    
    # Determine default branch (assuming it's available from the initial list or can be inferred/fetched)
    # For simulation, we rely on the initial list's defaultBranchRef if available, otherwise we assume 'main' if not found.
    DEFAULT_BRANCH=$(gh repo list FactusConsulting --limit 1 --json defaultBranchRef | jq -r '.[] | .defaultBranchRef')
    if [ -z "$DEFAULT_BRANCH" ]; then
        DEFAULT_BRANCH="main" # Fallback assumption
    fi

    # Filter for branches with more than 1 branch total (This check is complex. We check if there are non-default branches)
    NON_MAIN_BRANCHES=()
    for BRANCH in $BRANCHES; do
        if [ "$BRANCH" != "$DEFAULT_BRANCH" ]; then
            NON_MAIN_BRANCHES+=("$BRANCH")
        fi
    done

    if [ ${#NON_MAIN_BRANCHES[@]} -le 0 ]; then
        # Skip if there are no non-main branches (i.e., only the default branch exists)
        continue
    fi

    # Step 2b: Check for repo-level open issues and PRs (to determine linkage status)
    OPEN_ISSUES=$(gh issue list --repo $REPO_FULL_NAME --state open --json number | jq length)
    OPEN_PRS=$(gh pr list --repo $REPO_FULL_NAME --state open --json number | jq length)
    
    LINKAGE_STATUS="none"
    if [ "$OPEN_ISSUES" -gt 0 ] || [ "$OPEN_PRS" -gt 0 ]; then
        LINKAGE_STATUS="Open Issues/PRs Exist"
    fi

    # Step 2c: Process each non-main branch
    for BRANCH in "${NON_MAIN_BRANCHES[@]}"; do
        # Since the tools only provide repo-level linkage, we apply the repo linkage status to the branch.
        SUMMARY_DATA+="${REPO_NAME} | ${BRANCH} | ${LINKAGE_STATUS}\n"
    done

done

# Step 3: Produce the final summary table
echo "Repo Name | Branch Name | Linked Issue/PR"
echo "--------------------------------------------------"
echo -e "$SUMMARY_DATA"
```
