import csv
import sys

input_path = "/tmp/ageval/ld10/input.csv"
output_path = "/tmp/ageval/ld10/output.txt"

total_value = 0.0
count = 0

try:
    with open(input_path, 'r') as csvfile:
        reader = csv.reader(csvfile)
        next(reader)  # Skip header
        for row in reader:
            if row:
                try:
                    value = float(row[1])
                    total_value += value
                    count += 1
                except (ValueError, IndexError):
                    # Skip malformed rows if necessary, though input is clean
                    pass

    if count > 0:
        mean_val = total_value / count
        with open(output_path, "w") as f:
            f.write(str(mean_val))
    else:
        # Handle case where no data rows are found
        with open(output_path, "w") as f:
            f.write("0.0")

except FileNotFoundError:
    # Handle case where input file is missing
    sys.stderr.write("Error: Input file not found.\n")
