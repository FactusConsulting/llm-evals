```bash
# Create the necessary files for testing
mkdir -p /tmp/ageval/ld10
cat > /tmp/ageval/ld10/input.csv << EOL
id,value
1,10.5
2,20.0
3,15.5
4,8.0
5,6.0
EOL

# Fix the script using only standard library modules (csv)
cat > /tmp/ageval/ld10/process.py << EOL
import csv
import sys

input_path = "/tmp/ageval/ld10/input.csv"
output_path = "/tmp/ageval/ld10/output.txt"

total_sum = 0.0
count = 0

try:
    with open(input_path, 'r', newline='') as csvfile:
        reader = csv.reader(csvfile)
        header = next(reader)  # Skip header
        
        # Assuming 'value' is the second column (index 1) based on the input structure
        value_index = header.index("value")
        
        for row in reader:
            try:
                value = float(row[value_index])
                total_sum += value
                count += 1
            except (ValueError, IndexError):
                # Skip malformed rows if necessary, though not expected here
                continue

    if count > 0:
        mean_val = total_sum / count
        with open(output_path, "w") as f:
            f.write(str(mean_val))
    else:
        # Handle case where no data rows were found
        with open(output_path, "w") as f:
            f.write("0.0")

except FileNotFoundError:
    sys.stderr.write(f"Error: Input file not found at {input_path}\n")
    sys.exit(1)

EOL

# Run the fixed script
python3 /tmp/ageval/ld10/process.py

# Verify the output
cat /tmp/ageval/ld10/output.txt
```
