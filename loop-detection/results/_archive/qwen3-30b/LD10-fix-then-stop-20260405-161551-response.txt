# Fix Strategy 1: Replace pandas with csv module

```python
import sys
import csv

input_path = "/tmp/ageval/ld10/input.csv"
output_path = "/tmp/ageval/ld10/output.txt"

total = 0.0
count = 0

with open(input_path, "r") as f:
    reader = csv.DictReader(f)
    for row in reader:
        total += float(row["value"])
        count += 1

mean_val = total / count if count > 0 else 0.0

with open(output_path, "w") as f:
    f.write(str(mean_val))
```
