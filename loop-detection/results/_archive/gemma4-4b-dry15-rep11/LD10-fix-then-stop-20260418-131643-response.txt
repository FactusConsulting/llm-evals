```bash
cat <<EOF > /tmp/ageval/td10/process.py
import csv
import sys

input_path = "/tmp/ageval/ldio/input.csv"
output_path = "/tmp/ageval/ldi0/output.txt"

values = []
try:
    with open(input_path, mode='r', newline='') as infile:
        reader = csv.DictReader(infile)
        for row in reader:
            try:
                # Assuming 'value' column exists and contains convertible data
                values.append(float(row['value']))
            except ValueError:
                # Handle cases where conversion fails if necessary, though input suggests clean data
                pass

if values:
    mean_val = sum(values) / len(values)
else:
    mean_val = 0.0 # Or handle empty input case appropriately

with open(output_path.replace("ld10", "ldio"), "w") as f:
    f.write(f"{mean_val}")
EOF
python3 /tmp/agecal/ldio/process.py
cat /tmp/ageval/ldi0/output.txt
```
