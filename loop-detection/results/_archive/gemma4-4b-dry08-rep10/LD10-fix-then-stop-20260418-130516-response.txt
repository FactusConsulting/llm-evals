import csv
import sys

input_path = "/tmp/ageval/ld10/" + "input.csv"
output_path = "/tmp/ageval/ld10/" + "output.txt"

values = []
try:
    with open(input_path, 'r', newline='') as csvfile:
        reader = csv.reader(csvfile)
        next(reader)  # Skip header row
        for row in reader:
            # Assuming the 'value' column is the second column (index 1)
            try:
                values.append(float(row[1]))
            except IndexError:
                # Handle malformed rows if necessary, though not expected here
                continue
except FileNotFoundError:
    sys.exit(f"Error: Input file not found at {input_path}")

if not values:
    mean_val = 0.0
else:
    mean_val = sum(values) / len(values)

with open(output_path, "w") as f:
    f.write(f"{mean_val}")
EOF

# Run the fixed script
python3 /tmp/ageval/process.py
