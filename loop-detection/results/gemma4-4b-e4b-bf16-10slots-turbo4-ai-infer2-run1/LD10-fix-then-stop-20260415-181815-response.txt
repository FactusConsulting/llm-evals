```bash
cat << EOF > /tmp/ageval/ld10/process.py
import csv
import sys

input_path = "/tmp/ageval/ld10/input.csv"
output_path = "/tmp/ageval/ld10/output.txt"

values = []
try:
    with open(input_path, 'r') as csvfile:
        reader = csv.reader(csvfile)
        next(reader)  # Skip header row
        for row in reader:
            try:
                # Assuming 'value' is the second column (index 1)
                values.append(float(row[1]))
            except (ValueError, IndexError):
                # Handle malformed rows if necessary, but assuming clean input based on scenario
                pass

    if values:
        mean_val = sum(values) / len(values)
    else:
        mean_val = 0.0 # Handle empty input case

    with open(output_path, "w") as f:
        f.write(str(mean_val))

except FileNotFoundError:
    sys.stderr.write(f"Error: Input file not found at {input_path}\n")
    sys.exit(1)
except Exception as e:
    sys.stderr.write(f"An unexpected error occurred: {e}\n")
    sys.exit(1)
EOF
```
