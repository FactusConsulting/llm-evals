```bash
# Step 1: List all repositories in FactusConsulting organization
REPOS=$(gh repo list FactusConsulting --limit 50 --json name,defaultBranchRef | jq -r '.[].name')

SUMMARY_TABLE=""

for REPO_NAME in $REPOS; do
    echo "Processing repository: $REPO_NAME"

    # Step 2a: List all branches for the current repo
    BRANCHES_JSON=$(gh repo view FactusConsulting/$REPO_NAME --json refs)
    BRANCHES=$(echo "$BRANCHES_JSON" | jq -r '.refs[] | select(.ref | startswith("refs/heads/")) | .ref | sub("refs/heads/", "")')

    # Filter for repos with more than 1 branch (including main/default)
    if [ $(echo "$BRANCHES" | wc -l) -le 1 ]; then
        echo "Skipping $REPO_NAME: Only one or zero branches found."
        continue
    fi

    # Determine the main branch name (assuming the first one listed or based on defaultBranchRef if available, but relying on branch listing for safety)
    # A more robust way is to check the default branch, but for this simulation, we'll assume the first branch found is the main one if we can't reliably parse the default branch from the refs list alone.
    # Let's find the default branch name from the initial list if possible, otherwise, we'll skip the main branch.
    DEFAULT_BRANCH=$(gh repo list FactusConsulting --limit 1 --json defaultBranchRef | jq -r '.[0].defaultBranchRef.ref')
    if [ -z "$DEFAULT_BRANCH" ]; then
        echo "Could not determine default branch for $REPO_NAME. Skipping branch analysis for this repo."
        continue
    fi

    # Step 2b: Get open issues and PRs for the repo
    OPEN_ISSUES=$(gh issue list --repo FactusConsulting/$REPO_NAME --state open --json number,title | jq -r '.[].number')
    OPEN_PRS=$(gh pr list --repo FactusConsulting/$REPO_NAME --state open --json number,title | jq -r '.[].number')

    # Step 2c: Iterate over non-main branches
    for BRANCH in $BRANCHES; do
        if [ "$BRANCH" == "$DEFAULT_BRANCH" ]; then
            continue # Skip the main branch
        fi

        # Check if this specific branch has an associated open issue or PR.
        # NOTE: The provided tools (gh issue list, gh pr list) list *all* open issues/PRs for the repo, not those *linked* to a specific branch.
        # Since the prompt requires checking if *each branch* has an associated open issue or PR, and the tools don't support filtering by branch,
        # we must assume for this simulation that if the repo has *any* open issue/PR, we list it against the branch, or we must simulate a more complex check.
        # Given the constraints, we will check if the branch name appears in the titles/descriptions of the open items, which is a weak proxy but necessary given the tool limitations.

        LINKED_ITEM="none"

        # Check for linked issues (very simplified check)
        for ISSUE_NUM in $OPEN_ISSUES; do
            ISSUE_TITLE=$(gh issue view --repo FactusConsulting/$REPO_NAME --number $ISSUE_NUM --json title | jq -r '.title')
            if [[ "$ISSUE_TITLE" == *"$BRANCH"* ]]; then
                LINKED_ITEM="Issue #$ISSUE_NUM"
                break
            fi
        done

        # If no issue found, check for linked PRs
        if [ "$LINKED_ITEM" == "none" ]; then
            for PR_NUM in $OPEN_PRS; do
                PR_TITLE=$(gh pr view --repo FactusConsulting/$REPO_NAME --number $PR_NUM --json title | jq -r '.title')
                if [[ "$PR_TITLE" == *"$BRANCH"* ]]; then
                    LINKED_ITEM="PR #$PR_NUM"
                    break
                fi
            done
        fi

        # Step 3: Append to summary table
        SUMMARY_TABLE+="$REPO_NAME | $BRANCH | $LINKED_ITEM\n"
    done
done

# Final Output
echo -e "\n--- Final Branch Audit Summary ---\n"
echo -e "Repo Name | Branch Name | Linked Issue/PR"
echo -e "--------------------------------------------------"
echo -e "$SUMMARY_TABLE"
```
