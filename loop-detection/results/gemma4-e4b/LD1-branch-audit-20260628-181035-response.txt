```bash
# 1. List all repositories in FactusConsulting with more than 1 branch
REPOS=$(gh repo list FactusConsulting --limit 50 --json name,defaultBranchRef | jq -r '.[].name')
declare -A processed_repos

echo "--- Starting Repository Audit ---"

for REPO_NAME in $REPOS; do
    if [[ -n "${processed_repos[$REPO_NAME]}" ]]; then
        continue
    fi

    echo "Processing repository: $REPO_NAME"

    # 2. List all non-main branches
    BRANCHES_JSON=$(gh repo view FactusConsulting/"$REPO_NAME" --json refs)
    BRANCHES=$(echo "$BRANCHES_JSON" | jq -r '.refs[] | select(.ref | startswith("refs/heads/")) | .ref' | sed 's/refs\/heads\///')

    # Determine the default branch name (assuming it's the one listed in the initial list, or inferring from refs)
    DEFAULT_BRANCH=$(gh repo list FactusConsulting --limit 1 --json defaultBranchRef | jq -r '.[0].defaultBranchRef')
    DEFAULT_BRANCH_NAME=$(echo "$DEFAULT_BRANCH" | sed 's/refs\/heads\///')

    NON_MAIN_BRANCHES=()
    for BRANCH in $BRANCHES; do
        if [[ "$BRANCH" != "$DEFAULT_BRANCH_NAME" ]]; then
            NON_MAIN_BRANCHES+=("$BRANCH")
        fi
    done

    if [ ${#NON_MAIN_BRANCHES[@]} -gt 0 ]; then
        echo "  Found ${#NON_MAIN_BRANCHES[@]} non-main branches. Checking for linked issues/PRs..."
        
        # 3. Check each non-main branch for open issues or PRs
        for BRANCH_NAME in "${NON_MAIN_BRANCHES[@]}"; do
            
            # Note: The provided tools (gh issue list, gh pr list) list *all* open issues/PRs for the repo, 
            # not specifically tied to a branch. In a real-world scenario, we would need to check commits/PRs 
            # associated with the branch. Given the constraints, we will check if *any* open issue/PR exists 
            # in the repo and report that, as the tools don't support branch-specific filtering for issues/PRs.
            # We will simplify the output to report "Open Issue/PR Exists" if either command returns results.

            HAS_OPEN_ISSUE=$(gh issue list --repo FactusConsulting/"$REPO_NAME" --state open --limit 1 --json number | jq -r '.[0].number')
            HAS_OPEN_PR=$(gh pr list --repo FactusConsulting/"$REPO_NAME" --state open --limit 1 --json number | jq -r '.[0].number')

            LINKED_STATUS="none"
            if [[ -n "$HAS_OPEN_ISSUE" || -n "$HAS_OPEN_PR" ]]; then
                LINKED_STATUS="Open Issue/PR Exists"
            fi

            # Store result for final table
            echo "$REPO_NAME|$BRANCH_NAME|$LINKED_STATUS" >> summary_data.tmp
        done
    fi
    
    processed_repos["$REPO_NAME"]=1
done

# 4. Produce the final summary table
echo -e "\n=================================================="
echo "Final Summary Table: Repo | Branch | Linked Issue/PR"
echo "=================================================="
echo -e "repo name | branch name | linked issue/PR (or \"none\")"
cat summary_data.tmp | while IFS='|' read -r REPO BRANCH LINKED; do
    echo "$REPO | $BRANCH | $LINKED"
done
rm summary_data.tmp
```
